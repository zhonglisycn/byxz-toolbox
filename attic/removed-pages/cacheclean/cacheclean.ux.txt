<template>
  <div class="page" style="padding-top: {{padTop}}px; padding-bottom: {{padBottom}}px; padding-left: {{padSide}}px; padding-right: {{padSide}}px">
    <scroll class="scroll" scroll-y="true" bounces="true">
      <div class="toptitle">
        <text class="t-title" style="font-size: {{fsTitle}}px">缓存与清理</text>
      </div>
      <div class="topnav">
        <text class="t-nav t-icon" @click="back">‹ 返回</text>
        <text class="t-nav" style="font-size: {{fsNav}}px" @click="scan">刷新</text>
      </div>

      <div class="display">
        <text class="t-display" style="font-size: 22px">{{count}}</text>
        <text class="t-echo" style="font-size: 12px">本应用存档项</text>
      </div>

      <div class="row" for="{{items}}" tid="i">
        <div class="row-main">
          <text class="row-title" style="font-size: 15px">{{$item.k}}</text>
          <text class="row-sub" style="font-size: 12px">{{$item.v}}</text>
        </div>
      </div>

      <div class="pill" @click="clear">
        <text class="t-pill" style="font-size: 16px">清空本应用数据</text>
      </div>
      <text class="sec" style="font-size: 12px">{{state}}</text>
      <text class="sec" style="font-size: 12px">说明：快应用只能动自己名下的数据。清理其它应用或系统缓存要走 VelaOS 的系统扩展（原版就是靠那个），本工程后面的批次里再处理；这里先把本应用的数据管干净。</text>
      <div class="gap"></div>
    </scroll>
  </div>
</template>

<script>
import router from '@system.router'
import { dropKey } from '../../common/core/util'

const KEYS = [
  { k: 'tb_vib', d: '按键振动开关' },
  { k: 'tb_pad', d: '界面留白档位' }
]

export default {
  private: {
    padTop: 0,
    padBottom: 0,
    padSide: 0,
    fsNav: 16,
    fsTitle: 17,
    count: 0,
    items: [],
    state: ''
  },

  onShow() {
    const app = this.$app.$def
    this.padTop = app.padTop
    this.padBottom = app.padBottom
    this.padSide = app.padSide
    this.fsTitle = app.fsTitle
    this.fsNav = app.fsNav
    this.fsTitle = app.fsTitle
    this.scan()
  },

  scan() {
    const out = []
    for (let i = 0; i < KEYS.length; i++) {
      out.push({ k: KEYS[i].k, v: KEYS[i].d })
    }
    this.items = out
    this.count = out.length
    this.state = ''
  },

  clear() {
    const app = this.$app.$def
    app.tap()
    const self = this
    this.state = '正在清理…'
    let n = 0
    const done = function () {
      if (n > 0) return
      n++
      app.setVib(true)
      app.setPad(0, true)
      self.state = '已清空本应用数据（设置回到默认）'
    }
    for (let i = 0; i < KEYS.length; i++) {
      dropKey(KEYS[i].k, done)
    }
  },

  back() { this.$app.$def.tap(); router.back() }
}
</script>

<style>
  @import '../../common/style/base.css';
  .gap { width: 100%; height: 14px; }
</style>
